    COUNT(CASE WHEN xb = '女' THEN 1 END) AS female_count,  -- 统计女性教职工数量
    CAST(COUNT(CASE WHEN xb = '男' THEN 1 END) AS FLOAT) / COUNT(*) AS male_ratio,  -- 计算男性比例

    function generateChartOption(chartData) {
    const option = {
        title: {
            text: '学校论坛高频词',
            left: 'center'
        },
        tooltip: {
            show: true
        },
        series: [{
            type: 'wordCloud',
            size: ['80%', '80%'],
            rotationRange: [-90, 90],
            rotationStep: 45,
            gridSize: 20,
            shape: 'circle',
            textStyle: {
                fontFamily:'sans-serif',
                fontWeight: 'bold',
                color: function () {
                    return 'rgb(' + [
                        Math.round(Math.random() * 160),
                        Math.round(Math.random() * 160),
                        Math.round(Math.random() * 160)
                    ].join(',') + ')';
                }
            },
            emphasis: {
                textStyle: {
                    shadowBlur: 10,
                    shadowColor: '#333'
                }
            },
            data: chartData.data.map(item => ({ name: item.word, value: item.count }))
        }]
    };
    return option;
}

    function fetchDataAndUpdateChart() {
        const chartData = {
            "metadata": {

                    "count": 3000
                },
            ]
        };
        let option = generateChartOption(chartData);
        myChart.setOption(option);
    }

    // 初始加载数据
    fetchDataAndUpdateChart();

    document.getElementById('refreshBtn').addEventListener('click', fetchDataAndUpdateChart);
